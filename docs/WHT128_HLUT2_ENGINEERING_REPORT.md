$$\hat{w}_2(s, m) = (-1)^s \sigma_{r,g}\bigl(c_0 + m(c_1 - c_0)\bigr), \quad \hat{w}_3(s, m_2) = (-1)^s \sigma_{r,g} c_{m_2}, \quad H_{\text{bal}} = \frac{H_{\text{common}} + \lambda H_{\text{rare}}}{M_{\text{cal}}} + 0.015\,\bar{d}\,I_K, \quad \lambda = \frac{\operatorname{tr}(H_{\text{common}})}{\operatorname{tr}(H_{\text{rare}})}$$
